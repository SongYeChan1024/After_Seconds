System.Func<UnityEngine.Vector3[],float,bool,System.Collections.Generic.List<UnityEngine.Vector3>> Rounded=(points,radius,closed)=>{
 var result=new System.Collections.Generic.List<UnityEngine.Vector3>();
 if(!closed)result.Add(points[0]);

 for(int i=first;i<last;i++){
  var a=points[(i-1+points.Length)%points.Length];var b=points[i];var c=points[(i+1)%points.Length];
  var d0=(b-a).normalized;var d1=(c-b).normalized;
  float trim=UnityEngine.Mathf.Min(radius,UnityEngine.Vector3.Distance(a,b)*.27f,UnityEngine.Vector3.Distance(b,c)*.27f);
  if(UnityEngine.Vector3.Dot(d0,d1)>.999f){result.Add(b);continue;}

  for(int j=0;j<=8;j++){float t=j/8f;result.Add((1-t)*(1-t)*entry+2*(1-t)*t*b+t*t*exit);}
 }

};
