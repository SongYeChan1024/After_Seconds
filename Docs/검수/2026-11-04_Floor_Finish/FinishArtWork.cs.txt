// 손상된 시각 외형만 새 모델로 교체하고, 실제 씬의 앞뒤 모습을 렌더하여 검토한다.
// [코딩]
using System;using System.IO;using System.Linq;using System.Collections.Generic;
using UnityEngine;using UnityEngine.Rendering;using UnityEngine.Rendering.Universal;
using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;
using Newtonsoft.Json;using Newtonsoft.Json.Linq;
public static class FinishArtWork {
 public const string Root="Assets/Afrer_Seconds/";
 public static string W=>Path.GetFullPath(Path.Combine(Application.dataPath,"../../../tmp/B2_Pipes_20261104"));
 public static string F=>Path.GetFullPath(Path.Combine(Application.dataPath,"../../../tmp/1F_Facades_20261104"));
 public static Transform[] All(Scene s)=>s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();
 static string P(Transform t)=>t.parent==null?t.name:P(t.parent)+"/"+t.name;
 static float[] V(Vector3 v)=>new[]{v.x,v.y,v.z};
 static float[] Q(Quaternion q)=>new[]{q.x,q.y,q.z,q.w};
 public static Bounds B(GameObject go){var rr=go.GetComponentsInChildren<Renderer>(true).Where(r=>r.enabled&&r.gameObject.activeInHierarchy).ToArray();if(rr.Length==0)throw new Exception("No renderer: "+go.name);var b=rr[0].bounds;foreach(var r in rr.Skip(1))b.Encapsulate(r.bounds);return b;}
 static object ColliderData(Collider c){object shape=null;
 if(c is BoxCollider b)shape=new{center=V(b.center),size=V(b.size)};
 else if(c is CapsuleCollider a)shape=new{center=V(a.center),a.radius,a.height,a.direction};
 else if(c is SphereCollider z)shape=new{center=V(z.center),z.radius};
 else if(c is CharacterController cc)shape=new{center=V(cc.center),cc.height,cc.radius,cc.slopeLimit,cc.stepOffset};
 else if(c is MeshCollider m)shape=new{mesh=AssetDatabase.GetAssetPath(m.sharedMesh),meshName=m.sharedMesh?m.sharedMesh.name:null,m.convex};
 return new{path=P(c.transform),type=c.GetType().Name,pos=V(c.transform.position),rot=Q(c.transform.rotation),scale=V(c.transform.lossyScale),c.enabled,c.isTrigger,shape};}
 public static JArray Protected(){var result=new JArray();foreach(var path in AssetDatabase.FindAssets("t:Scene",new[]{Root+"01.Scenes/Stage"}).Select(AssetDatabase.GUIDToAssetPath).OrderBy(x=>x)){
 var s=EditorSceneManager.OpenPreviewScene(path);try{var all=All(s);result.Add(JObject.FromObject(new{scene=path,colliders=all.SelectMany(t=>t.GetComponents<Collider>()).Select(ColliderData).OrderBy(x=>JsonConvert.SerializeObject(x)).ToArray(),poses=all.Select(t=>new{path=P(t),pos=V(t.localPosition),rot=Q(t.localRotation),scale=V(t.localScale)}).OrderBy(x=>x.path).ThenBy(x=>JsonConvert.SerializeObject(x)).ToArray()}));}finally{EditorSceneManager.ClosePreviewScene(s);}}
 return result;}
 public static string Capture(){if(EditorApplication.isPlaying||Enumerable.Range(0,SceneManager.sceneCount).Any(i=>SceneManager.GetSceneAt(i).isDirty))throw new Exception("Clean edit scenes required");var p=Path.Combine(W,"protected_before.json");if(File.Exists(p))throw new Exception("Already captured");File.WriteAllText(p,Protected().ToString());return "Six floor collider and pose baselines captured from preview scenes";}
 public static void Open1F(){if(Enumerable.Range(0,SceneManager.sceneCount).Any(i=>SceneManager.GetSceneAt(i).isDirty))throw new Exception("Unsaved scene; do not switch");EditorSceneManager.OpenScene(Root+"01.Scenes/Stage/Level_1F_Checkpoint.unity");}
 public static string Import1F(){var manifest=JArray.Parse(File.ReadAllText(Path.Combine(F,"model/facades_manifest.json")));var results=new JArray();
 foreach(var item in manifest){string name=(string)item["name"],dir=Root+"05.Model/Environment/1F/",modelPath=dir+name+"_Rebuilt.fbx",prefabPath=dir+name+".prefab";
 if(File.Exists(modelPath))throw new Exception("Derivative already exists "+name);File.Copy((string)item["fbx"],modelPath);AssetDatabase.ImportAsset(modelPath,ImportAssetOptions.ForceSynchronousImport);
 var imp=(ModelImporter)AssetImporter.GetAtPath(modelPath);imp.importAnimation=false;imp.importCameras=false;imp.importLights=false;imp.addCollider=false;imp.materialImportMode=ModelImporterMaterialImportMode.ImportStandard;imp.materialLocation=ModelImporterMaterialLocation.InPrefab;imp.importNormals=ModelImporterNormals.Import;imp.importTangents=ModelImporterTangents.None;imp.SaveAndReimport();
 var map=new Dictionary<string,Material>();foreach(var slot in item["materials"]){var slotName=(string)slot["name"];string matPath=dir+"Materials/Facade_"+slotName+".mat";var mat=AssetDatabase.LoadAssetAtPath<Material>(matPath);if(!mat){mat=new Material(Shader.Find("Universal Render Pipeline/Lit"));mat.name="Facade_"+slotName;var rgba=slot["baseColor"].ToObject<float[]>();mat.SetColor("_BaseColor",new Color(rgba[0],rgba[1],rgba[2],rgba[3]));mat.SetFloat("_Metallic",(float)slot["metallic"]);mat.SetFloat("_Smoothness",1-(float)slot["roughness"]);AssetDatabase.CreateAsset(mat,matPath);}map.Add(slotName,mat);}
 string guid=AssetDatabase.AssetPathToGUID(prefabPath);var contents=PrefabUtility.LoadPrefabContents(prefabPath);try{var before=B(contents);var colliders=contents.GetComponentsInChildren<Collider>(true).Length;foreach(var old in contents.GetComponentsInChildren<Renderer>(true))old.enabled=false;
 var model=(GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>(modelPath),contents.scene);model.name="Rebuilt_Visual";model.transform.SetParent(contents.transform,false);
 foreach(var renderer in model.GetComponentsInChildren<Renderer>(true)){renderer.sharedMaterials=renderer.sharedMaterials.Select(m=>map[map.Keys.First(k=>m.name.StartsWith(k,StringComparison.Ordinal))]).ToArray();renderer.shadowCastingMode=ShadowCastingMode.On;renderer.receiveShadows=true;}
 var bounds=B(contents);if((bounds.size-before.size).magnitude>.003f||(bounds.center-before.center).magnitude>.003f)throw new Exception("Bounds/axis mismatch "+name+" before "+before+" after "+bounds);
 if(contents.GetComponentsInChildren<Collider>(true).Length!=colliders)throw new Exception("Collider changed");PrefabUtility.SaveAsPrefabAsset(contents,prefabPath);results.Add(JObject.FromObject(new{name,prefabPath,guid,modelPath,oldSize=V(before.size),newSize=V(bounds.size),oldCenter=V(before.center),newCenter=V(bounds.center),colliders}));
 }finally{PrefabUtility.UnloadPrefabContents(contents);}if(guid!=AssetDatabase.AssetPathToGUID(prefabPath))throw new Exception("Prefab GUID changed");
 item["modelAssetPath"]=modelPath;item["prefabAssetPath"]=prefabPath;item["prefabGuidPreserved"]=guid;item["originalModelPreserved"]=true;item["sceneInstancesUnmoved"]=true;File.WriteAllText(Root+"06.Data/AssetSources/"+name+"_Rebuilt_source_manifest.json",item.ToString());}
 AssetDatabase.SaveAssets();File.WriteAllText(Path.Combine(F,"unity_import.json"),results.ToString());return results.ToString();}
 public static string Review1F(){var s=SceneManager.GetActiveScene();if(!s.path.EndsWith("Level_1F_Checkpoint.unity"))throw new Exception("1F must be active");var all=All(s);var targets=all.Where(t=>t.name=="LoadingBay_Shutter"||t.name=="Exterior_Architecture_Corner"||t.name.Contains("LoadingBay")||t.name.Contains("ExteriorCorner")).Select(t=>new{path=P(t),pos=V(t.position),rot=Q(t.rotation),scale=V(t.lossyScale)}).ToArray();File.WriteAllText(Path.Combine(F,"scene_targets.json"),JsonConvert.SerializeObject(targets,Formatting.Indented));return JsonConvert.SerializeObject(targets);}
 public static void Render(string file,Vector3 eye,Vector3 target,bool hideCeiling=false){var s=SceneManager.GetActiveScene();var hidden=new List<Renderer>();var all=All(s);foreach(var cc in all.SelectMany(t=>t.GetComponents<CharacterController>()))foreach(var r in cc.GetComponentsInChildren<Renderer>(true))if(r.enabled){r.enabled=false;hidden.Add(r);}if(hideCeiling)foreach(var t in all.Where(t=>t.name.IndexOf("Ceiling",StringComparison.OrdinalIgnoreCase)>=0))foreach(var r in t.GetComponentsInChildren<Renderer>())if(r.enabled&&!hidden.Contains(r)){r.enabled=false;hidden.Add(r);}
 var g=new GameObject("Temporary_FinishArtReviewCamera");g.hideFlags=HideFlags.HideAndDontSave;var c=g.AddComponent<Camera>();c.enabled=false;c.transform.SetPositionAndRotation(eye,Quaternion.LookRotation(target-eye));c.fieldOfView=62;c.nearClipPlane=.04f;c.farClipPlane=150;c.useOcclusionCulling=false;c.allowHDR=true;c.GetUniversalAdditionalCameraData().renderPostProcessing=true;var rt=new RenderTexture(1600,1000,24);var prev=RenderTexture.active;try{c.targetTexture=rt;c.Render();c.Render();c.Render();RenderTexture.active=rt;var image=new Texture2D(1600,1000,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,1600,1000),0,0);image.Apply();File.WriteAllBytes(file,image.EncodeToPNG());UnityEngine.Object.DestroyImmediate(image);}finally{c.targetTexture=null;RenderTexture.active=prev;rt.Release();UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(g);foreach(var r in hidden)r.enabled=true;}}
 public static string Photos1F(string phase){Render(Path.Combine(F,phase+"_front.png"),new Vector3(0,3,-27.7f),new Vector3(-1,1.65f,-24.8f));Render(Path.Combine(F,phase+"_rear.png"),new Vector3(0,3,-22.3f),new Vector3(0,1.4f,-25.3f));return "1F review photos captured";}
 public static string Verify(){var before=JArray.Parse(File.ReadAllText(Path.Combine(W,"protected_before.json")));var after=JArray.Parse(Protected().ToString());var result=new JArray();foreach(var old in before){var now=after.First(x=>(string)x["scene"]==(string)old["scene"]);bool col=JToken.DeepEquals(old["colliders"],now["colliders"]);var oldPoses=old["poses"].ToArray();var nowPoses=now["poses"].ToArray();var missing=oldPoses.Where(x=>!nowPoses.Any(y=>JToken.DeepEquals(x,y))).ToArray();result.Add(JObject.FromObject(new{scene=(string)old["scene"],collidersPreserved=col,changedOriginalPoses=missing}));if(!col||missing.Length>0){File.WriteAllText(Path.Combine(W,"verification.json"),result.ToString());throw new Exception("Protected data mismatch "+old["scene"]);}}
 File.WriteAllText(Path.Combine(W,"verification.json"),result.ToString());return "All six floor colliders and original transforms preserved";}
}
