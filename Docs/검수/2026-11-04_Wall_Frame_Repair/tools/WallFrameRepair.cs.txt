 static string MeshKey(Mesh m){string guid;long id;AssetDatabase.TryGetGUIDAndLocalFileIdentifier(m,out guid,out id);return AssetDatabase.GetAssetPath(m)+"|"+id+"|"+m.name;}
 static bool Architecture(MeshFilter mf)
 {
  var r=mf.GetComponent<Renderer>();if(!Visible(r)||!mf.sharedMesh)return false;
  var a=AssetDatabase.GetAssetPath(mf.sharedMesh);var p=P(mf.transform);var n=mf.name;
  if(a.Contains("/Props/")||a.Contains("_BevelSmooth")||System.Text.RegularExpressions.Regex.IsMatch(Path.GetFileName(a),@"^U\d{2}_")||a.Contains("ObservationPod")||a.Contains("DualFoodHatch")||a.Contains("DomePendant"))return false;
  if(r.sharedMaterials.Any(m=>m&&m.HasProperty("_Surface")&&m.GetFloat("_Surface")>0))return false;
  return p.Contains("/Geometry/")||p.Contains("RebuiltArchitectureVisual")||p.Contains("AS_CurrentArchitecture_VisualOnly")||System.Text.RegularExpressions.Regex.IsMatch(n,"Wall|Ceiling|Floor|Door|Frame|Deck|Landing|Stair|Pen_Retaining",System.Text.RegularExpressions.RegexOptions.IgnoreCase);
 }
 static object MeshData(Mesh m)=>new{vertices=m.vertices.Select(V).ToArray(),normals=m.normals.Select(V).ToArray(),uv=m.uv.Select(v=>new[]{v.x,v.y}).ToArray(),uv2=m.uv2.Select(v=>new[]{v.x,v.y}).ToArray(),tangents=m.tangents.Select(t=>new[]{t.x,t.y,t.z,t.w}).ToArray(),submeshes=Enumerable.Range(0,m.subMeshCount).Select(i=>m.GetTriangles(i)).ToArray(),source=AssetDatabase.GetAssetPath(m),m.name};
 static object ColliderShape(Collider c)
 {
  object shape=null;if(c is BoxCollider b)shape=new{center=V(b.center),size=V(b.size)};else if(c is SphereCollider s)shape=new{center=V(s.center),s.radius};else if(c is CapsuleCollider a)shape=new{center=V(a.center),a.radius,a.height,a.direction};else if(c is CharacterController cc)shape=new{center=V(cc.center),cc.radius,cc.height,cc.slopeLimit,cc.stepOffset};else if(c is MeshCollider mc)shape=new{mesh=mc.sharedMesh?MeshKey(mc.sharedMesh):null,mc.convex,mc.cookingOptions};
  return new{path=P(c.transform),type=c.GetType().Name,c.enabled,c.isTrigger,c.contactOffset,material=AssetDatabase.GetAssetPath(c.sharedMaterial),shape};
 }
 static JObject Guard(Scene s)=>JObject.FromObject(new{scene=s.path,poses=All(s).Select(t=>new{path=P(t),pos=V(t.localPosition),rot=Q(t.localRotation),scale=V(t.localScale),active=t.gameObject.activeSelf,layer=t.gameObject.layer}).OrderBy(x=>x.path).ToArray(),colliders=All(s).SelectMany(t=>t.GetComponents<Collider>()).Select(ColliderShape).OrderBy(x=>JsonConvert.SerializeObject(x)).ToArray(),behaviours=All(s).SelectMany(t=>t.GetComponents<MonoBehaviour>()).Select(b=>new{path=b?P(b.transform):"missing",type=b?b.GetType().FullName:"missing"}).OrderBy(x=>x.path).ThenBy(x=>x.type).ToArray()});
 static Bounds B(Transform t){var rr=t.GetComponentsInChildren<Renderer>(true).Where(Visible).ToArray();if(rr.Length==0)return new Bounds(t.position,Vector3.one);var b=rr[0].bounds;foreach(var r in rr.Skip(1))b.Encapsulate(r.bounds);return b;}
 static void Render(Scene scene,JToken shot,string phase)
 {
  var e=shot["eye"].ToObject<float[]>();var t=shot["target"].ToObject<float[]>();var hidden=new List<Renderer>();foreach(var cc in All(scene).SelectMany(x=>x.GetComponents<CharacterController>()))foreach(var r in cc.GetComponentsInChildren<Renderer>(true).Where(Visible)){r.enabled=false;hidden.Add(r);}
  var go=new GameObject("Temporary_WallFrameReviewCamera");go.hideFlags=HideFlags.HideAndDontSave;var c=go.AddComponent<Camera>();c.enabled=false;c.transform.SetPositionAndRotation(new Vector3(e[0],e[1],e[2]),Quaternion.LookRotation(new Vector3(t[0]-e[0],t[1]-e[1],t[2]-e[2])));c.fieldOfView=62;c.nearClipPlane=.035f;c.farClipPlane=200;c.useOcclusionCulling=false;c.allowHDR=true;c.GetUniversalAdditionalCameraData().renderPostProcessing=true;c.GetUniversalAdditionalCameraData().volumeLayerMask=~0;
  var rt=new RenderTexture(1280,800,24);var prev=RenderTexture.active;try{c.targetTexture=rt;c.Render();c.Render();RenderTexture.active=rt;var image=new Texture2D(1280,800,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,1280,800),0,0);image.Apply();File.WriteAllBytes(Path.Combine(Out,(string)shot["name"]+"_"+phase+".png"),image.EncodeToPNG());UnityEngine.Object.DestroyImmediate(image);}finally{c.targetTexture=null;RenderTexture.active=prev;rt.Release();UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(go);foreach(var r in hidden)r.enabled=true;}
 }
 static JArray Shots(Scene s,Transform[] all)
 {
  var shots=new JArray();var rooms=all.Where(t=>System.Text.RegularExpressions.Regex.IsMatch(t.name,@"^Room\d{2}_(Isolation|CoolingPool|ObservationCorridor|ControlDeck|SampleStorage|DecontaminationAirlock|CentrifugeHall|CentralControl|SubjectChamber|ExitCheckpoint)$")).ToArray();int i=0;
  foreach(var room in rooms){var b=B(room);for(int side=0;side<2;side++){float sign=side==0?1:-1;var eye=new Vector3(b.center.x-b.extents.x*.55f*sign,room.position.y+1.65f,b.center.z+b.extents.z*.6f*sign);if(room.name=="Room04_ControlDeck"&&side==1)eye=new Vector3(2,4.15f,-25);var target=b.center;target.y=room.position.y+1.5f;shots.Add(JObject.FromObject(new{scene=s.path,name=Floor(s.path)+"_Room_"+(i++).ToString("D2"),kind="room",owner=P(room),eye=V(eye),target=V(target)}));}}
  var doors=all.Where(t=>t.name.IndexOf("Door",StringComparison.OrdinalIgnoreCase)>=0&&t.GetComponentsInChildren<MeshFilter>(true).Any(Architecture)).Where(t=>!Ancestors(t).Any(p=>p.name.IndexOf("Door",StringComparison.OrdinalIgnoreCase)>=0)).ToArray();i=0;
  foreach(var door in doors){var b=B(door);var axis=b.size.x<b.size.z?Vector3.right:Vector3.forward;float distance=Mathf.Clamp(Mathf.Max(b.size.x,b.size.z)*.7f,1.8f,3.5f);for(int side=0;side<2;side++){var center=b.center;center.y=b.min.y+Mathf.Min(1.5f,b.size.y*.5f);var eye=center+axis*distance*(side==0?1:-1);eye.y=b.min.y+Mathf.Min(1.65f,b.size.y*.55f);shots.Add(JObject.FromObject(new{scene=s.path,name=Floor(s.path)+"_Door_"+(i++).ToString("D2"),kind="door",owner=P(door),eye=V(eye),target=V(center)}));}}
  return shots;
 }
 static IEnumerable<Transform> Ancestors(Transform t){for(var p=t.parent;p;p=p.parent)yield return p;}

 static void ValidateDerivedMeshes(){var result=new List<object>();foreach(var g in AssetDatabase.FindAssets("t:Mesh",new[]{Root+"05.Model/Environment"})){var path=AssetDatabase.GUIDToAssetPath(g);if(!Path.GetFileName(path).StartsWith("WF_"))continue;var m=AssetDatabase.LoadAssetAtPath<Mesh>(path);var v=m.vertices;var indices=m.triangles;bool ok=v.All(x=>float.IsFinite(x.x)&&float.IsFinite(x.y)&&float.IsFinite(x.z))&&m.tangents.All(t=>float.IsFinite(t.x)&&float.IsFinite(t.y)&&float.IsFinite(t.z)&&float.IsFinite(t.w))&&indices.All(x=>x>=0&&x<v.Length)&&indices.Length%3==0&&m.normals.Length==v.Length&&(m.uv.Length==0||m.uv.Length==v.Length)&&(m.uv2.Length==0||m.uv2.Length==v.Length);result.Add(new{asset=path,valid=ok,vertices=v.Length,triangles=indices.Length/3,uv2=m.uv2.Length,emptyVisual=indices.Length==0});if(!ok)throw new Exception("Invalid derived mesh "+path);}Save(Path.Combine(Out,"mesh_validation.json"),result);}
 static void ExportCurrentGeometry(){var geometry=new Dictionary<string,object>();var instances=new List<object>();int index=0;foreach(var path in Stages()){var s=EditorSceneManager.OpenScene(path,OpenSceneMode.Single);foreach(var mf in All(s).SelectMany(t=>t.GetComponents<MeshFilter>()).Where(Architecture)){string key=MeshKey(mf.sharedMesh);if(!geometry.ContainsKey(key))geometry[key]=MeshData(mf.sharedMesh);var m=mf.transform.localToWorldMatrix;instances.Add(new{id=(index++).ToString("D4"),scene=path,floor=Floor(path),path=P(mf.transform),mesh=key,matrix=Enumerable.Range(0,16).Select(i=>m[i/4,i%4]).ToArray()});}}Save(Path.Combine(Temp,"geometry_after.json"),geometry);Save(Path.Combine(Temp,"instances_after.json"),instances);}
