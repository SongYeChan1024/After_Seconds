var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();
var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<UnityEngine.Transform>(true)).ToArray();
var pod=all.FirstOrDefault(t=>t.name=="ObservationPod");
return new{scene=s.path,dirty=s.isDirty,playing=UnityEditor.EditorApplication.isPlaying,selection=UnityEditor.Selection.transforms.Select(t=>new{t.name,pos=t.position,localScale=t.localScale,worldScale=t.lossyScale}).ToArray(),pod=pod==null?null:new{name=pod.name,pos=pod.position,localPos=pod.localPosition,scale=pod.localScale,worldScale=pod.lossyScale,bounds=pod.GetComponentsInChildren<UnityEngine.Renderer>(true).Where(r=>r.enabled).Select(r=>new{r.name,center=r.bounds.center,size=r.bounds.size}).ToArray()},rooms=all.Where(t=>t.name=="Room01_Isolation").Select(t=>new{t.name,pos=t.position,scale=t.localScale}).ToArray(),props=all.Where(t=>t.name.StartsWith("P01_")||t.name.StartsWith("P02_")||t.name.Contains("MonitoringConsole")||t.name=="Cell_Lamp").Select(t=>new{t.name,pos=t.position,localPos=t.localPosition,scale=t.localScale}).ToArray()};
